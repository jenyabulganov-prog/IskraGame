  function updateBoss(dt){
    if(!boss||boss.dead)return;boss.flash=Math.max(0,boss.flash-dt);boss.timer-=dt;
    if(boss.mode==='idle'&&boss.timer<=0){boss.dir=player.x<boss.x?-1:1;if(Math.random()<.52){boss.mode='charge';boss.timer=1.15;boss.vx=boss.dir*82}else{boss.mode='jump';boss.timer=1.25;boss.vy=-205;boss.vx=boss.dir*48;boss.onGround=false}}
    if(boss.mode==='charge'){boss.x+=boss.vx*dt;if(boss.x<590){boss.x=590;boss.dir=1;boss.vx=Math.abs(boss.vx)}if(boss.x>835){boss.x=835;boss.dir=-1;boss.vx=-Math.abs(boss.vx)}if(boss.timer<=0){boss.mode='idle';boss.timer=.8;boss.vx=0}}
    else if(boss.mode==='jump'){boss.vy+=520*dt;boss.x+=boss.vx*dt;boss.y+=boss.vy*dt;if(boss.y+boss.h>=GROUND_Y){boss.y=GROUND_Y-boss.h;boss.vy=0;boss.vx=0;boss.onGround=true;boss.mode='idle';boss.timer=.75;dustBurst(boss.x+boss.w/2,GROUND_Y,0,18);shake=.14}}
    const bh={x:boss.x,y:boss.y,w:boss.w,h:boss.h};if(aabb(player,bh)&&boss.flash<=0){const stomp=player.vy>40&&player.y+player.h-5<boss.y+10;if(stomp){boss.hp--;boss.flash=.6;player.vy=-185;score+=800;burst(boss.x+boss.w/2,boss.y+8,'#ffce66',18);sparkleBurst(boss.x+boss.w/2,boss.y+6,'#fff',8);audio.sfx('bossHit');shake=.15;toast(`БОСС: ${boss.hp}/${boss.maxHp}`,'#ffe39a');if(boss.hp<=0){boss.dead=true;boss.vx=0;score+=3000;setTimeout(()=>{if(state==='play'){unlockedLevel=3;saveProgress();audio.sfx('win');setState('bosswin')}},900)}}else if(invuln<=0)damage(false)}
  }

  }
