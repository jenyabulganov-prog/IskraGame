  function resetCanvas(){ctx.setTransform(1,0,0,1,0,0);ctx.fillStyle='#111';ctx.fillRect(0,0,canvas.width,canvas.height);ctx.setTransform(SCALE,0,0,SCALE,0,0);ctx.imageSmoothingEnabled=true;ctx.lineCap='round';ctx.lineJoin='round'}
  function linear(x1,y1,x2,y2,a,b,c=null){const g=ctx.createLinearGradient(x1,y1,x2,y2);g.addColorStop(0,a);if(c){g.addColorStop(.55,b);g.addColorStop(1,c)}else g.addColorStop(1,b);return g}
  function radial(x,y,r,a,b){const g=ctx.createRadialGradient(x-r*.25,y-r*.3,r*.12,x,y,r);g.addColorStop(0,a);g.addColorStop(1,b);return g}
  function rr(x,y,w,h,r,fill,stroke=null,lw=.7){ctx.beginPath();ctx.roundRect(x,y,w,h,r);ctx.fillStyle=fill;ctx.fill();if(stroke){ctx.strokeStyle=stroke;ctx.lineWidth=lw;ctx.stroke()}}
  function ell(cx,cy,rx,ry,fill,stroke=null,lw=.7){ctx.beginPath();ctx.ellipse(cx,cy,rx,ry,0,0,Math.PI*2);ctx.fillStyle=fill;ctx.fill();if(stroke){ctx.strokeStyle=stroke;ctx.lineWidth=lw;ctx.stroke()}}
  function line(x1,y1,x2,y2,color,w=1){ctx.beginPath();ctx.moveTo(x1,y1);ctx.lineTo(x2,y2);ctx.strokeStyle=color;ctx.lineWidth=w;ctx.stroke()}

    ctx.fillStyle=linear(0,0,0,H,sky[0],sky[1],sky[2]);ctx.fillRect(0,0,W,H);
    ctx.globalAlpha=.22;for(let y=18;y<150;y+=26){ctx.fillStyle='#fff';ctx.fillRect(0,y,W,.7)}ctx.globalAlpha=1;
    if(theme==='forest')drawForestLayers(cam);else if(theme==='cliffs')drawCliffLayers(cam);else if(theme==='boss')drawBossBackdrop(cam);else drawMeadowLayers(cam);
  }
